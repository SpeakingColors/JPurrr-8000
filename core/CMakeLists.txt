file(GLOB JPURRR_CORE_SRC CONFIGURE_DEPENDS src/*.cpp src/dsp/*.cpp)
add_library(jpurrr_core STATIC ${JPURRR_CORE_SRC})

if(JPURRR_SANITIZE)

endif()

  target_compile_options(jpurrr_core PRIVATE -Wall -Wextra)
endif()
